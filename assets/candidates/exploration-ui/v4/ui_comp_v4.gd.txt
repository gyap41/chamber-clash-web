extends "res://.local/ui_comp_v3.gd"
const OUT4 := "res://assets/candidates/exploration-ui/v4/"
var skins: Dictionary = {}
func texture_file(path: String) -> Texture2D:
	var im := Image.load_from_file(path)
	assert(im != null)
	var lo := Vector2i(im.get_width(),im.get_height())
	var hi := Vector2i(-1,-1)
	# Ignore almost transparent generation residue outside the actual material.
	for y in range(im.get_height()):
		for x in range(im.get_width()):
			var p: Color = im.get_pixel(x,y)
			if p.a < .5: continue
			lo = lo.min(Vector2i(x,y))
			hi = hi.max(Vector2i(x,y))
	assert(hi.x >= lo.x)
	print("Material bounds: ",path.get_file()," ",Rect2i(lo,hi-lo+Vector2i.ONE))
	return ImageTexture.create_from_image(im.get_region(Rect2i(lo,hi-lo+Vector2i.ONE)))
func skin(name: String) -> Texture2D:
	if not skins.has(name): skins[name] = texture_file(OUT4+name+".png")
	return skins[name]
func tile(tex: Texture2D, source: Rect2, target: Rect2, tint: Color = Color.WHITE) -> void:
	var atlas := AtlasTexture.new()
	atlas.atlas = tex
	atlas.region = source
	var n := TextureRect.new()
	n.texture = atlas
	n.expand_mode = TextureRect.EXPAND_IGNORE_SIZE
	n.stretch_mode = TextureRect.STRETCH_SCALE
	n.position = target.position
	n.modulate = tint
	page.add_child(n)
	n.set_deferred("size",target.size)
func nine(name: String, target: Rect2, edge: float = 9, tint: Color = Color.WHITE) -> void:
	var tex := skin(name)
	var size := tex.get_size()
	var fraction: float = .12 if name == "cell" else .10
	var src_x: Array = [0.0,size.x*fraction,size.x*(1.0-fraction),size.x]
	var src_y: Array = [0.0,size.y*fraction,size.y*(1.0-fraction),size.y]
	var dst_x: Array = [target.position.x,target.position.x+edge,target.end.x-edge,target.end.x]
	var dst_y: Array = [target.position.y,target.position.y+edge,target.end.y-edge,target.end.y]
	for y in range(3):
		for x in range(3):
			tile(tex,Rect2(src_x[x],src_y[y],src_x[x+1]-src_x[x],src_y[y+1]-src_y[y]),Rect2(dst_x[x],dst_y[y],dst_x[x+1]-dst_x[x],dst_y[y+1]-dst_y[y]),tint)
func box(rect: Rect2, color: Color, border: Color = Color("4c6668")) -> void:
	if rect.size == Vector2(57,57):
		nine("cell",rect,5,Color.WHITE if color.r > .1 else Color(.42,.49,.48))
	elif rect.size == Vector2(step,step):
		var tex := skin("cell")
		tile(tex,Rect2(tex.get_size()*.15,tex.get_size()*.7),rect,Color(.87,.95,.9))
	elif rect.size == Vector2(61,48):
		nine("cell",rect,6)
	elif rect.size == Vector2(507,240):
		nine("detail",rect,12)
	else: super.box(rect,color,border)
func segment(a: Vector2,b: Vector2,color: Color,width: float = 2) -> void:
	if width == 1:
		super.segment(a,b,color,width)
		return
	var tex := skin("cell")
	var size := tex.get_size()
	if is_equal_approx(a.x,b.x):
		tile(tex,Rect2(0,size.y*.15,size.x*.1,size.y*.7),Rect2(a-Vector2(2,0),Vector2(5,b.y-a.y)),Color(1.12,1.08,.94))
	else:
		tile(tex,Rect2(size.x*.15,0,size.x*.7,size.y*.1),Rect2(a-Vector2(0,2),Vector2(b.x-a.x,5)),Color(1.12,1.08,.94))
	# State outline remains distinct from the cloth texture.
	super.segment(a,b,Color(color,.65),1)
func button_state(text: String,rect: Rect2,state: String,primary: bool = false) -> void:
	var tint := Color.WHITE
	if state == "hover": tint = Color(1.25,1.18,1.05)
	if state == "disabled": tint = Color(.43,.48,.46)
	if state == "pressed":
		tint = Color(.73,.79,.73)
		rect.position.y += 2
	if primary and state == "normal": tint = Color(1.16,1.08,.86)
	nine("button",rect,9,tint)
	var n := label(text,rect,18,Color("ecdfc4") if state != "disabled" else Color("8e9a95"))
	n.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	n.vertical_alignment = VERTICAL_ALIGNMENT_CENTER
func button(text: String,rect: Rect2,primary: bool = false) -> void:
	if rect.position.y == 641: rect.position.y = 648
	button_state(text,rect,"normal",primary)

func capture4(name: String) -> void:
	await process_frame
	await RenderingServer.frame_post_draw
	assert(root.get_texture().get_image().save_png(OUT4+name+".png") == OK)
	page.free()
func run() -> void:

	game = load("res://scenes/game/exploration.tscn").instantiate()
	game.random_floor = true
	game.authored_campaign = true
	root.add_child(game)
	game.set_physics_process(false)
	var shop_id: String = game.floor_data.rooms.keys().filter(func(id): return game.floor_data.rooms[id].role == "shop")[0]
	game.move_to_room(shop_id,game.room_data(shop_id).doors[0].arrival)
	game.exploration.inventory.gold[0] = 32
	game.refresh_hud()
	await process_frame
	variant = 1
	bag()
	await capture4("bag")
	variant = 2
	shop()
	await capture4("shop")
	states()
	await capture4("states")
	print("PASS: textured UI comps and four button states rendered; shared inventory checks passed")
	quit()
